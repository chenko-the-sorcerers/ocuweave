cmake_minimum_required(VERSION 3.20)
project(ocuweave VERSION 0.12.0 LANGUAGES CXX)
add_library(ocuweave_core STATIC native/src/evidence.cpp native/src/segmentation.cpp)
target_compile_features(ocuweave_core PUBLIC cxx_std_17)
target_include_directories(ocuweave_core PUBLIC native/include)
set_target_properties(ocuweave_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(ocuweave_native SHARED native/bindings/c_api.cpp)
target_link_libraries(ocuweave_native PRIVATE ocuweave_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(ocuweave_native_test tests/native/test_evidence.cpp)
  target_link_libraries(ocuweave_native_test PRIVATE ocuweave_core)
  add_test(NAME evidence_cpu COMMAND ocuweave_native_test)
endif()
if(BUILD_TESTING)
  add_executable(ocuweave_segmentation_test tests/native/test_segmentation.cpp)
  target_link_libraries(ocuweave_segmentation_test PRIVATE ocuweave_core)
  add_test(NAME segmentation_cpu COMMAND ocuweave_segmentation_test)
endif()

# Framework-free CPU reference engine. No LibTorch, BLAS, or Python dependency.
add_library(ocuweave_nn STATIC native/src/nn/tensor.cpp native/src/nn/lqnet.cpp native/src/nn/backend.cpp)
target_compile_features(ocuweave_nn PUBLIC cxx_std_17)
target_include_directories(ocuweave_nn PUBLIC native/include)
if(MSVC)
  target_compile_options(ocuweave_nn PRIVATE /W4)
else()
  target_compile_options(ocuweave_nn PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endif()
if(BUILD_TESTING)
  add_executable(ocuweave_nn_test tests/native/nn/test_tensor.cpp)
  target_link_libraries(ocuweave_nn_test PRIVATE ocuweave_nn)
  add_test(NAME native_nn_operators COMMAND ocuweave_nn_test)
  add_executable(ocuweave_lqnet_test tests/native/nn/test_lqnet.cpp)
  target_link_libraries(ocuweave_lqnet_test PRIVATE ocuweave_nn)
  add_test(NAME native_lqnet_training COMMAND ocuweave_lqnet_test ${CMAKE_CURRENT_BINARY_DIR}/native-test.ckpt)
endif()

option(OCUWEAVE_ENABLE_CUDA "Enable custom convolution/matmul CUDA kernels" OFF)
if(OCUWEAVE_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(ocuweave_nn PRIVATE native/cuda/nn_kernels.cu)
  target_compile_definitions(ocuweave_nn PUBLIC OCUWEAVE_CUDA=1)
  target_link_libraries(ocuweave_nn PRIVATE CUDA::cudart)
  set_target_properties(ocuweave_nn PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()
target_sources(ocuweave_nn PRIVATE native/src/nn/io.cpp)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE OCUWEAVE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT OCUWEAVE_COMMIT)
  set(OCUWEAVE_COMMIT "unknown")
endif()
execute_process(COMMAND git status --porcelain WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE OCUWEAVE_STATUS ERROR_QUIET)
if(OCUWEAVE_STATUS)
  set(OCUWEAVE_DIRTY "true")
else()
  set(OCUWEAVE_DIRTY "false")
endif()
add_executable(ocuweave-lqnet native/tools/lqnet_main.cpp)
target_link_libraries(ocuweave-lqnet PRIVATE ocuweave_nn)
target_compile_definitions(ocuweave-lqnet PRIVATE OCUWEAVE_VERSION="${PROJECT_VERSION}" OCUWEAVE_COMMIT="${OCUWEAVE_COMMIT}" OCUWEAVE_DIRTY="${OCUWEAVE_DIRTY}")
add_executable(ocuweave-predict native/tools/predict_main.cpp)
target_link_libraries(ocuweave-predict PRIVATE ocuweave_nn)
if(BUILD_TESTING)
  add_executable(ocuweave_nn_io_test tests/native/nn/test_io.cpp)
  target_link_libraries(ocuweave_nn_io_test PRIVATE ocuweave_nn)
  add_test(NAME native_io_metrics COMMAND ocuweave_nn_io_test ${CMAKE_CURRENT_BINARY_DIR}/batch-test.bin)
  if(OCUWEAVE_ENABLE_CUDA)
    add_executable(ocuweave_cuda_test tests/native/nn/test_cuda.cpp)
    target_link_libraries(ocuweave_cuda_test PRIVATE ocuweave_nn)
    add_test(NAME native_cuda_parity COMMAND ocuweave_cuda_test)
  endif()
endif()
