cmake_minimum_required(VERSION 3.20)
project(ocuweave VERSION 0.1.0 LANGUAGES CXX)
add_library(ocuweave_core STATIC native/src/evidence.cpp)
target_compile_features(ocuweave_core PUBLIC cxx_std_17)
target_include_directories(ocuweave_core PUBLIC native/include)
set_target_properties(ocuweave_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(ocuweave_native SHARED native/bindings/c_api.cpp)
target_link_libraries(ocuweave_native PRIVATE ocuweave_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(ocuweave_native_test tests/native/test_evidence.cpp)
  target_link_libraries(ocuweave_native_test PRIVATE ocuweave_core)
  add_test(NAME evidence_cpu COMMAND ocuweave_native_test)
endif()
